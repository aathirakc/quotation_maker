function calculateCartTotal() { let total = 0; // 1. Grab all the product rows inside the cart const cartItems = document.querySelectorAll('.cart-item'); // 2. Loop through each item row cartItems.forEach(item => { // Extract price from the data attribute and convert it to a float number const priceElement = item.querySelector('.product-price'); const price = parseFloat(priceElement.getAttribute('data-price')); // Extract quantity from the input box const quantityElement = item.querySelector('.product-quantity'); const quantity = parseInt(quantityElement.value) || 0; // 3. Update the total text formatted to 2 decimal places document.getElementById('total-amount').innerText = `$${total.toFixed(2)}`; } // 5. Setup Listeners to recalculate whenever quantities change document.querySelectorAll('.product-quantity').forEach(input => { input.addEventListener('change', calculateCartTotal); input.addEventListener('keyup', calculateCartTotal); // Handles manual typing }); // Run once on page load to set the initial total calculateCartTotal();
